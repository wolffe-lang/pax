# px103: the typed sudo session. Section headers [root], [wolf], [lobo] name who types.
# {OUT} is the capture file; each line is typed as: LINE >> {OUT} 2>&1; echo rc=$? >> {OUT}
# [root] is typed at the root console; [wolf] after `su wolf` (uid 1000, the installer's account);
# [lobo] at the root console, every line run as lobo by `sudo -n -u lobo /bin/sudo -n ...` (lobo's
# shell is /bin/false, so lobo cannot type): its lines are the arguments after `/bin/sudo -n`.
[root]
id
sudo -n /bin/id
sudo -n -u lobo /bin/id
sudo -n -u wolf /bin/id
sudo -n -g lobo /bin/id
sudo -n /bin/sh -c 'exit 7'
sudo -n -u nosuchuser /bin/id
sudo -n -l /bin/id
[wolf]
id
sudo -n /bin/id
sudo -n -u lobo /bin/id
sudo -n -g lobo /bin/id
sudo -n /bin/sh -c 'exit 7'
sudo -n -u nosuchuser /bin/id
sudo -n -l /bin/id
sudo -n -l /bin/false
sudo -k
[lobo]
/bin/id
-l /bin/id
